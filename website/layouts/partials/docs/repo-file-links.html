{{- /* Turns inline code that is a real repository path into a link to that path on the latest main branch. Fenced code is and is left alone. An existing anchor is left alone, so a link that already points at main is not wrapped a second time. Pass dict "content" "page". */ -}} {{- $html := .content -}} {{- $page := .page -}} {{- $anchors := slice -}} {{- range $i, $anchor := findRE `(?s)` $html 5000 -}} {{- $token := printf "TRUXANCHOR%dTOKEN" $i -}} {{- $anchors = $anchors | append (dict "token" $token "text" $anchor) -}} {{- $html = replace $html $anchor $token 1 -}} {{- end -}} {{- $pat := printf `(%s)` (partial "docs/repo-path-re.html" .) -}} {{- range findRE $pat $html 5000 -}} {{- $raw := replaceRE `^|$` "" . -}} {{- $url := partial "docs/repo-file-url.html" (dict "path" $raw "page" $page) -}} {{- if $url -}} {{- $html = replace $html (printf "%s" $raw) (printf `%s` $url $raw) -}} {{- end -}} {{- end -}} {{- range $anchors -}} {{- $html = replace $html .token .text 1 -}} {{- end -}} {{- $out := $html | safeHTML -}} {{- return $out -}}
